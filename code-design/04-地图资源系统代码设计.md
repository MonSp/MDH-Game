## 2. 核心类设计

### 2.1 ResourceManager

```typescript
class ResourceManager {
  private static instance: ResourceManager;
  private activeResources: Map<string, ResourceNode>;
  private resourcePool: ObjectPool<ResourceNode>;
  private readonly RESPAWN_CHANCE = 0.7;
  private readonly SPAWN_RADIUS = 50;

  static getInstance(): ResourceManager {
    if (!ResourceManager.instance) {
      ResourceManager.instance = new ResourceManager();
    }
    return ResourceManager.instance;
  }

  initialize(mapWidth: number, mapHeight: number, initialCount: number): void {
    this.activeResources = new Map();
    this.resourcePool = new ObjectPool<>(100, () => this.createResourceNode());

    for (let i = 0; i < initialCount; i++) {
      this.spawnRandomResource(mapWidth, mapHeight);
    }
  }

  private createResourceNode(): ResourceNode {
    return {
      id: `resource_${Date.now()}_${Math.random()}`,
      type: ResourceType.SpiritField,
      x: 0,
      y: 0,
      respawnTimer: 0
    };
  }

  spawnRandomResource(mapWidth: number, mapHeight: number): ResourceNode | null {
    const types = Object.keys(ResourceType) as ResourceType[];
    const weights = [0.5, 0.35, 0.15];
    const type = this.weightedRandom(types, weights);

    const node = this.resourcePool.acquire();
    node.id = `resource_${Date.now()}_${Math.random()}`;
    node.type = type;
    node.x = Math.random() * mapWidth;
    node.y = Math.random() * mapHeight;
    node.lastCollectedAt = undefined;

    this.activeResources.set(node.id, node);
    return node;
  }

  collectResource(playerId: string, resourceId: string): CollectResult {
    const resource = this.activeResources.get(resourceId);
    if (!resource) {
      return { success: false, reason: 'resource_not_found' };
    }

    const player = PlayerService.getPlayer(playerId);
    const config = RESOURCE_CONFIGS[resource.type];

    if (config.cultivationReward > 0) {
      player.addCultivation(config.cultivationReward);
    }
    if (config.spiritStoneReward > 0) {
      player.addSpiritStones(config.spiritStoneReward);
    }

    let specialDrop: string | null = null;
    if (config.specialDropChance && Math.random() < config.specialDropChance) {
      specialDrop = config.specialDropItem!;
      player.addItem(specialDrop);
    }

    resource.lastCollectedAt = Date.now();
    this.activeResources.delete(resourceId);
    this.resourcePool.release(resource);

    this.scheduleRespawn(resource, player.x, player.y);

    return { success: true, rewards: { cultivation: config.cultivationReward, spiritStones: config.spiritStoneReward, specialDrop } };
  }

  private scheduleRespawn(collectedResource: ResourceNode, playerX: number, playerY: number): void {
    if (Math.random() >= this.RESPAWN_CHANCE) {
      return;
    }

    const offsetX = (Math.random() - 0.5) * 2 * this.SPAWN_RADIUS;
    const offsetY = (Math.random() - 0.5) * 2 * this.SPAWN_RADIUS;
    const newX = Math.max(0, Math.min(this.mapWidth, playerX + offsetX));
    const newY = Math.max(0, Math.min(this.mapHeight, playerY + offsetY));

    setTimeout(() => {
      this.spawnAtPosition(collectedResource.type, newX, newY);
    }, 30000);
  }

  private spawnAtPosition(type: ResourceType, x: number, y: number): void {
    const node = this.resourcePool.acquire();
    node.id = `resource_${Date.now()}_${Math.random()}`;
    node.type = type;
    node.x = x;
    node.y = y;
    node.lastCollectedAt = undefined;

    this.activeResources.set(node.id, node);
  }

  private weightedRandom<T>(items: T[], weights: number[]): T {
    const total = weights.reduce((a, b) => a + b, 0);
    let random = Math.random() * total;
    for (let i = 0; i < items.length; i++) {
      random -= weights[i];
      if (random <= 0) return items[i];
    }
    return items[items.length - 1];
  }

  getActiveResources(): ResourceNode[] {
    return Array.from(this.activeResources.values());
  }
}
```

### 2.3 NPCResourceCompetition

```typescript
class NPCResourceCompetition {
  private npcResources: Map<string, Set<string>>;

  constructor() {
    this.npcResources = new Map();
  }

  npcCollects(npcId: string, resourceId: string): void {
    if (!this.npcResources.has(npcId)) {
      this.npcResources.set(npcId, new Set());
    }
    this.npcResources.get(npcId)!.add(resourceId);
  }

  canNpcCollect(npcId: string, resourceId: string): boolean {
    for (const [npc, resources] of this.npcResources.entries()) {
      if (npc !== npcId && resources.has(resourceId)) {
        return false;
      }
    }
    return true;
  }

  npcReleases(npcId: string, resourceId: string): void {
    this.npcResources.get(npcId)?.delete(resourceId);
  }
}
```

  x INTEGER NOT NULL,

### 3.2 采集记录表

```sql
CREATE TABLE collection_records (
  id TEXT PRIMARY KEY,
  player_id TEXT NOT NULL,
  resource_id TEXT NOT NULL,
  collected_at INTEGER NOT NULL,
  rewards_cultivation INTEGER DEFAULT 0,
  rewards_spirit_stones INTEGER DEFAULT 0,
  special_drop TEXT,
  FOREIGN KEY (player_id) REFERENCES players(id),
  FOREIGN KEY (resource_id) REFERENCES resource_nodes(id)
);
```

## 5. 使用示例

```typescript
const resourceManager = ResourceManager.getInstance();
resourceManager.initialize(1000, 1000, 50);

EventBus.on('player:collect', ({ playerId, resourceId }) => {
  const result = resourceManager.collectResource(playerId, resourceId);
  if (result.success) {
    SendRewardNotification(playerId, result.rewards);
    BroadcastToArea(player.areaId, {
      type: 'resource:update',
      action: 'despawn',
      resourceId
    });
  }
});

EventBus.on('npc:collect', ({ npcId, resourceId }) => {
  if (!npcResourceCompetition.canNpcCollect(npcId, resourceId)) {
    return;
  }
  resourceManager.collectResource(npcId, resourceId);
  npcResourceCompetition.npcCollects(npcId, resourceId);
});
```
