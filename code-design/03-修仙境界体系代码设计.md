## 2. 核心类设计

### 2.1 CultivationService

```typescript
class CultivationService {
  private static instance: CultivationService;

  static getInstance(): CultivationService {
    if (!CultivationService.instance) {
      CultivationService.instance = new CultivationService();
    }
    return CultivationService.instance;
  }

  getRealmConfig(realm: CultivationRealm): RealmConfig {
    return REALM_CONFIGS[realm];
  }

  getNextRealm(currentRealm: CultivationRealm): CultivationRealm | null {
    if (currentRealm === CultivationRealm.Tribulation) {
      return null;
    }
    return currentRealm + 1;
  }

  canBreakthrough(player: Player): boolean {
    const currentConfig = this.getRealmConfig(player.realm);
    return player.cultivation >= currentConfig.requiredCultivation;
  }

  attemptBreakthrough(player: Player): BreakthroughResult {
    if (!this.canBreakthrough(player)) {
      return { success: false, reason: 'cultivation_insufficient' };
    }

    const currentConfig = this.getRealmConfig(player.realm);
    if (player.spiritStones < currentConfig.spiritStoneCost) {
      return { success: false, reason: 'spirit_stones_insufficient' };
    }

    const nextRealm = this.getNextRealm(player.realm);
    if (!nextRealm) {
      return { success: false, reason: 'max_realm_reached' };
    }

    player.spiritStones -= currentConfig.spiritStoneCost;
    player.realm = nextRealm;
    this.applyRealmBonus(player, nextRealm);

    return { success: true, newRealm: nextRealm };
  }

  applyRealmBonus(player: Player, realm: CultivationRealm): void {
    const config = this.getRealmConfig(realm);
    player.maxHealth *= config.healthMultiplier;
    player.maxSpirit *= config.spiritMultiplier;
    player.attack *= config.powerMultiplier;
    player.currentHealth = player.maxHealth;
    player.currentSpirit = player.maxSpirit;
  }
}
```

### 2.3 PlayerRealmState

```typescript
class PlayerRealmState {
  constructor(
    public realm: CultivationRealm,
    public cultivation: number,
    public maxCultivation: number
  ) {}

  addCultivation(amount: number): void {
    this.cultivation += amount;
    if (this.cultivation > this.maxCultivation) {
      this.cultivation = this.maxCultivation;
    }
  }

  canAttemptBreakthrough(): boolean {
    return this.cultivation >= this.maxCultivation;
  }
}
```

## 3. 数据库设计

### 3.1 境界表

```sql
CREATE TABLE realms (
  id INTEGER PRIMARY KEY,
  name TEXT NOT NULL,
  required_cultivation INTEGER NOT NULL,
  spirit_stone_cost INTEGER NOT NULL,
  health_multiplier REAL DEFAULT 1.0,
  spirit_multiplier REAL DEFAULT 1.0,
  power_multiplier REAL DEFAULT 1.0
);
```

### 3.2 玩家境界表

```sql
CREATE TABLE player_realm (
  player_id TEXT PRIMARY KEY,
  realm INTEGER NOT NULL DEFAULT 1,
  cultivation INTEGER NOT NULL DEFAULT 0,
  FOREIGN KEY (player_id) REFERENCES players(id),
  FOREIGN KEY (realm) REFERENCES realms(id)
);
```

// server/game/services/CultivationPlanningLLM.ts

interface CultivationPlanningContext {
  npc_id: string;
  current_realm: CultivationRealm;
  target_realm: CultivationRealm;
  cultivation_resources: {
    spirit_stones: number;
    soul_building_pills: number;
    high_quality_stones: number;
  };
  personality: NPCPersonality;
  family_status: {
    treasury: number;
    members_count: number;
  };
  world_context: {
    at_war_with: string[];
    family_stability: number;
  };
}

class CultivationPlanningLLM {
  async generatePlan(context: CultivationPlanningContext): Promise<CultivationPlan> {
    const prompt = this.buildPlanningPrompt(context);
    const response = await LLMGatewayService.getInstance().call(prompt);
    return JSON.parse(response);
  }

  private buildPlanningPrompt(context: CultivationPlanningContext): string {
    return `
      你是一个修仙世界的修士，当前境界为${REALM_CONFIGS[context.current_realm].name}。
      目标突破至${REALM_CONFIGS[context.target_realm].name}。
      个人资源：${context.cultivation_resources.spirit_stones}灵石。
      家族状态：国库${context.family_status.treasury}灵石，成员${context.family_status.members_count}人。
      请生成一个7天的修炼突破计划，包括闭关、资源准备、时机选择等。
      返回JSON格式的计划。
    `;
  }
}

interface CultivationPlan {
  plan_id: string;
  horizon_days: number;
  primary_goal: string;
  sub_tasks: Array<{
    task_id: number;
    description: string;
    priority: number;
    target_completion_day: number;
    action_type: 'CULTIVATE_BREAKTHROUGH' | 'STATE_AFFAIRS';
    action_params: Record<string, any>;
  }>;
}
