```typescript
class PlayerService {
  private static instance: PlayerService;
  private players: Map<string, Player>;
  private playerDataCache: Map<string, PlayerData>;

  static getInstance(): PlayerService {
    if (!PlayerService.instance) {
      PlayerService.instance = new PlayerService();
    }
    return PlayerService.instance;
  }

  createPlayer(id: string, name: string): Player {
    const country = CountryService.getInstance().getRandomCountry();
    const familyInfo = FamilyService.getInstance().assignPlayerToFamily(id, country);

    const player = new Player({
      id,
      name,
      country,
      familyId: familyInfo.familyId,
      realm: CultivationRealm.Mortal,
      cultivation: 0,
      attributes: {
        health: 100,
        maxHealth: 100,
        spirit: 100,
        maxSpirit: 100,
        attack: 10,
        defense: 5,
        moveSpeed: 100,
        spiritStone: 0
      },
      position: { x: 0, y: 0 },
      state: PlayerState.Idle
    });

    player.applyCountryTrait();
    this.players.set(id, player);
    this.savePlayerData(id);

    return player;
  }

  getPlayer(id: string): Player | undefined {
    return this.players.get(id);
  }

  removePlayer(id: string): void {
    this.players.delete(id);
    this.playerDataCache.delete(id);
  }

  private async savePlayerData(playerId: string): Promise<void> {
    const player = this.players.get(playerId);
    if (!player) return;

    const data: PlayerData = player.toData();
    await DataService.getInstance().savePlayer(data);
  }

  private async loadPlayerData(playerId: string): Promise<PlayerData | null> {
    return await DataService.getInstance().loadPlayer(playerId);
  }

  getOnlinePlayers(): Player[] {
    return Array.from(this.players.values());
  }

  getPlayersInArea(areaId: string): Player[] {
    return this.getOnlinePlayers().filter(p => p.areaId === areaId);
  }
}
```

### 2.2 Player

```typescript
class Player {
  readonly id: string;
  name: string;
  country: Country;
  familyId: string;
  realm: CultivationRealm;
  cultivation: number;
  areaId: string;

  health: number;
  maxHealth: number;
  spirit: number;
  maxSpirit: number;
  attack: number;
  defense: number;
  baseMoveSpeed: number;
  spiritStones: number;

  position: { x: number; y: number };
  state: PlayerState;
  stateTimer: number;

  private eventBus: typeof EventBus;

  constructor(data: PlayerData) {
    this.id = data.id;
    this.name = data.name;
    this.country = data.country;
    this.familyId = data.familyId;
    this.realm = data.realm;
    this.cultivation = data.cultivation;
    this.health = data.attributes.health;
    this.maxHealth = data.attributes.maxHealth;
    this.spirit = data.attributes.spirit;
    this.maxSpirit = data.attributes.maxSpirit;
    this.attack = data.attributes.attack;
    this.defense = data.attributes.defense;
    this.baseMoveSpeed = data.attributes.moveSpeed;
    this.spiritStones = data.attributes.spiritStone;
    this.position = { ...data.position };
    this.state = data.state;
    this.areaId = '';
    this.stateTimer = 0;
    this.eventBus = EventBus;
  }

  applyCountryTrait(): void {
    const trait = CountryService.getInstance().getCountryTrait(this.country);
    switch (trait.type) {
      case 'battle_exp':
        this.attack *= (1 + trait.value / 100);
        break;
      case 'spirit_cap':
        this.maxSpirit *= (1 + trait.value / 100);
        this.spirit = this.maxSpirit;
        break;
      case 'move_speed':
        this.baseMoveSpeed *= (1 + trait.value / 100);
        break;
      case 'spirit_absorption':
        break;
    }
  }

  update(deltaTime: number): void {
    this.stateTimer += deltaTime;

    switch (this.state) {
      case PlayerState.Idle:
        this.recoverSpirit(deltaTime);
        break;
      case PlayerState.Sitting:
        this.recoverSpirit(deltaTime * 3);
        break;
      case PlayerState.Moving:
        break;
      case PlayerState.Fighting:
        break;
    }
  }

  private recoverSpirit(deltaTime: number): void {
    const recoveryRate = 1;
    this.spirit = Math.min(this.maxSpirit, this.spirit + recoveryRate * (deltaTime / 1000));
  }

  moveTo(x: number, y: number): void {
    this.position.x = x;
    this.position.y = y;
    this.setState(PlayerState.Moving);
  }

  sit(): void {
    this.setState(PlayerState.Sitting);
  }

  stand(): void {
    this.setState(PlayerState.Idle);
  }

  takeDamage(amount: number, attackerId: string): void {
    const actualDamage = Math.max(1, amount - this.defense);
    this.health -= actualDamage;

    this.eventBus.emit(PlayerEvent.HEALTH_CHANGED, {
      playerId: this.id,
      health: this.health,
      maxHealth: this.maxHealth
    });

    if (this.health <= 0) {
      this.onDeath(attackerId);
    }
  }

  private onDeath(killerId: string): void {
    this.setState(PlayerState.Dead);
    this.eventBus.emit(PlayerEvent.PLAYER_DIED, {
      playerId: this.id,
      killerId
    });

    setTimeout(() => {
      this.respawn();
    }, 5000);
  }

  private respawn(): void {
    const familyConfig = FamilyService.getInstance().getFamilyConfig(this.familyId);
    const capital = CountryService.getInstance().getCountryConfig(familyConfig.country).capitalPosition;

    this.position.x = capital.x;
    this.position.y = capital.y;
    this.health = this.maxHealth;
    this.spirit = this.maxSpirit;
    this.setState(PlayerState.Idle);

    this.eventBus.emit(PlayerEvent.PLAYER_RESPAWNED, { playerId: this.id });
  }

  setState(newState: PlayerState): void {
    if (this.state !== newState) {
      const oldState = this.state;
      this.state = newState;
      this.stateTimer = 0;
      this.eventBus.emit(PlayerEvent.STATE_CHANGED, {
        playerId: this.id,
        from: oldState,
        to: newState
      });
    }
  }

  addCultivation(amount: number): void {
    const realmConfig = CultivationService.getInstance().getRealmConfig(this.realm);
    this.cultivation += amount;

    if (this.cultivation >= realmConfig.requiredCultivation) {
      this.eventBus.emit(PlayerEvent.CULTIVATION_FULL, { playerId: this.id });
    }

    this.eventBus.emit(PlayerEvent.CULTIVATION_GAINED, {
      playerId: this.id,
      amount,
      total: this.cultivation
    });
  }

  addSpiritStones(amount: number): void {
    this.spiritStones += amount;
    this.eventBus.emit(PlayerEvent.SPIRIT_STONES_CHANGED, {
      playerId: this.id,
      amount: this.spiritStones
    });
  }

  toData(): PlayerData {
    return {
      id: this.id,
      name: this.name,
      country: this.country,
      familyId: this.familyId,
      realm: this.realm,
      cultivation: this.cultivation,
      attributes: {
        health: this.health,
        maxHealth: this.maxHealth,
        spirit: this.spirit,
        maxSpirit: this.maxSpirit,
        attack: this.attack,
        defense: this.defense,
        moveSpeed: this.baseMoveSpeed,
        spiritStone: this.spiritStones
      },
      position: { ...this.position },
      state: this.state,
      createdAt: 0,
      lastLoginAt: Date.now()
    };
  }
}
```

### 2.3 PlayerInputHandler

```typescript
class PlayerInputHandler {
  private player: Player;
  private inputQueue: InputCommand[];

  constructor(player: Player) {
    this.player = player;
    this.inputQueue = [];
  }

  handleInput(input: InputCommand): void {
    switch (input.type) {
      case InputType.Move:
        this.player.moveTo(input.x, input.y);
        break;
      case InputType.Sit:
        this.player.sit();
        break;
      case InputType.Stand:
        this.player.stand();
        break;
      case InputType.Attack:
        this.handleAttack(input.targetId);
        break;
      case InputType.Interact:
        this.handleInteract(input.targetId, input.action);
        break;
    }
  }

  private handleAttack(targetId: string): void {
    const target = GameWorld.getInstance().getNPC(targetId) || PlayerService.getPlayer(targetId);
    if (!target) return;

    const distance = this.player.position.distanceTo(target.position);
    if (distance > this.player.attackRange) return;

    this.player.setState(PlayerState.Fighting);
    EventBus.emit(PlayerEvent.ATTACK, {
      attackerId: this.player.id,
      targetId,
      damage: this.player.attack
    });
  }

  private handleInteract(targetId: string, action: InteractionAction): void {
    InteractionSystem.getInstance().interact(this.player.id, targetId, action);
  }
}

enum InputType {
  Move = 'move',
  Sit = 'sit',
  Stand = 'stand',
  Attack = 'attack',
  Interact = 'interact'
}

interface InputCommand {
  type: InputType;
  x?: number;
  y?: number;
  targetId?: string;
  action?: InteractionAction;
}
```

```typescript
interface PlayerInteractionContext {
  npc_id: string;
  npc_name: string;
  npc_position: string;
  npc_cultivation: string;
  personality: NPCPersonality;
  player_interaction_history: {
    total_interactions: number;
    gift_count: number;
    attack_count: number;
    trade_count: number;
    relationship_level: number;
  };
  current_world_state: {
    at_war_with: string[];
    player_country: string;
    player_realm: string;
  };
}

class PlayerInteractionLLM {
  async generateResponse(context: PlayerInteractionContext): Promise<InteractionResponse> {
    const prompt = this.buildInteractionPrompt(context);
    const response = await LLMGatewayService.getInstance().call(prompt);
    return JSON.parse(response);
  }

  private buildInteractionPrompt(context: PlayerInteractionContext): string {
    return `
      你是一个修仙世界的高阶NPC，名字是${context.npc_name}。
      你的人设：野心${context.personality.ambition}，谨慎${context.personality.caution}，
      忠诚${context.personality.loyalty}，贪婪${context.personality.greed}。
      玩家与你的交互历史：交互${context.player_interaction_history.total_interactions}次，
      送礼${context.player_interaction_history.gift_count}次，
      攻击${context.player_interaction_history.attack_count}次。
      当前关系等级：${context.player_interaction_history.relationship_level}。
      请生成一段符合你人设的对话回复，返回JSON格式。
    `;
  }
}

interface InteractionResponse {
  npc_id: string;
  response_type: 'DIALOGUE' | 'TRADE' | 'REJECT';
  dialogue_content: {
    text: string;
    emotion: string;
    action_hint: string;
    follow_up_options: string[];
  };
  relation_adjustment: number;
}
```
