```typescript
interface CountryConfig {
```

## 2. 核心类设计

### 2.1 CountryService

```typescript
class CountryService {
  private static instance: CountryService;

  static getInstance(): CountryService {
    if (!CountryService.instance) {
      CountryService.instance = new CountryService();
    }
    return CountryService.instance;
  }

  getRandomCountry(): Country {
    const countries = Object.keys(Country) as Country[];
    return countries[Math.floor(Math.random() * countries.length)];
  }

  getCountryConfig(country: Country): CountryConfig {
    return COUNTRY_CONFIGS[country];
  }

  getCountryTrait(country: Country): CountryConfig['trait'] {
    return COUNTRY_CONFIGS[country].trait;
  }

  getAllCountries(): Country[] {
    return Object.keys(Country) as Country[];
  }
}
```

### 2.2 CountryEntity

```typescript
class CountryEntity {
  readonly id: Country;
  readonly config: CountryConfig;

  constructor(country: Country) {
    this.id = country;
    this.config = COUNTRY_CONFIGS[country];
  }

  getName(): string {
    return this.config.name;
  }

  getTrait(): CountryConfig['trait'] {
    return this.config.trait;
  }

  getCapitalPosition(): { x: number; y: number } {
    return { ...this.config.capitalPosition };
  }
}
```

```sql
CREATE TABLE countries (
  id TEXT PRIMARY KEY,
  name TEXT NOT NULL,
  culture TEXT,
  trait_type TEXT,
  trait_value INTEGER,
  capital_x INTEGER,
  capital_y INTEGER
);
```

### 3.2 玩家国家关联表

```sql
CREATE TABLE player_country (
  player_id TEXT PRIMARY KEY,
  country_id TEXT NOT NULL,
  joined_at INTEGER NOT NULL,
  FOREIGN KEY (country_id) REFERENCES countries(id)
);
```

## 5. 使用示例

```typescript
// 获取随机国家用于玩家出生
const countryService = CountryService.getInstance();

const trait = countryService.getCountryTrait(playerCountry);

// 应用国家特质加成
if (trait.type === 'battle_exp') {
  player.battleExpRate *= (1 + trait.value / 100);
```

## 6. LLM决策集成

### 6.1 国家战略LLM调用

```typescript
// server/game/services/NationalStrategyLLM.ts

interface NationalStrategyContext {
  country: Country;
  national_status: {
    territory: number;
    total_families: number;
    allied_countries: string[];
    at_war_with: string[];
    treasury: number;
    military_strength: number;
  };
  world_context: {
    current_season: string;
    major_events: string[];
  };
}

class NationalStrategyLLM {
  async generateStrategy(context: NationalStrategyContext): Promise<NationalStrategyPlan> {
    const prompt = this.buildStrategyPrompt(context);
    const response = await LLMGatewayService.getInstance().call(prompt);
    return JSON.parse(response);
  }

  private buildStrategyPrompt(context: NationalStrategyContext): string {
    return `
      你是一个修仙世界的"天道意志"，负责为${CountryService.getName(context.country)}制定战略计划。
      当前状态：与${context.national_status.at_war_with.join(', ')}交战，国库${context.national_status.treasury}灵石。
      请生成一个月的战略计划，包括军事、外交、资源等方面的任务。
      返回JSON格式的战略计划。
    `;
  }
}

interface NationalStrategyPlan {
  plan_id: string;
  horizon_days: number;
  primary_goal: string;
  sub_tasks: Array<{
    task_id: number;
    description: string;
    priority: number;
    target_completion_day: number;
    action_type: 'MILITARY_ORDER' | 'DIPLOMACY' | 'CULTURAL_STRATEGY';
    action_params: Record<string, any>;
  }>;
}
```
